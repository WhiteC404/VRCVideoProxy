cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(VRCVideoProxy VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "VRCVideoProxy builds on Windows only.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(vrcvp_core STATIC src/core.cpp)
target_include_directories(vrcvp_core PUBLIC src)
target_compile_definitions(vrcvp_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00 VRCVP_VERSION=L\"${PROJECT_VERSION}\")
target_link_libraries(vrcvp_core PUBLIC winhttp shell32)
if(MSVC)
  add_compile_options(/W4 /utf-8 /permissive-)
  target_compile_options(vrcvp_core PRIVATE /W4 /utf-8 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -finput-charset=UTF-8)
  target_compile_options(vrcvp_core PRIVATE -Wall -Wextra -Wpedantic -finput-charset=UTF-8)
  add_link_options(-static -static-libgcc -static-libstdc++ -municode)
endif()
add_executable(VRCVideoProxy WIN32 src/main.cpp src/settings.cpp src/resources.rc)
target_include_directories(VRCVideoProxy PRIVATE src)
if(MSVC)
  target_link_options(VRCVideoProxy PRIVATE /MANIFEST:NO)
endif()
target_link_libraries(VRCVideoProxy PRIVATE vrcvp_core user32 gdi32 comctl32)
include(CTest)
if(BUILD_TESTING)
  add_executable(VRCVideoProxyProbe tests/probe.cpp)
  target_link_libraries(VRCVideoProxyProbe PRIVATE vrcvp_core)
  add_executable(VRCVideoProxyTests tests/tests.cpp)
  target_link_libraries(VRCVideoProxyTests PRIVATE vrcvp_core)
  add_test(NAME core_and_launch_contract COMMAND VRCVideoProxyTests $<TARGET_FILE:VRCVideoProxy> $<TARGET_FILE:VRCVideoProxyProbe>)
  add_test(NAME release_script_contract COMMAND ${CMAKE_COMMAND}
    "-DSOURCE_ROOT=${CMAKE_CURRENT_SOURCE_DIR}"
    "-DTEST_ROOT=${CMAKE_CURRENT_BINARY_DIR}"
    "-DVERSION=${PROJECT_VERSION}"
    "-DGH_PROBE=$<TARGET_FILE:VRCVideoProxyProbe>"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/release_script.cmake")
endif()
install(TARGETS VRCVideoProxy RUNTIME DESTINATION .)
install(FILES config/VRCVideoProxy.ini README.md README.zh-CN.md CONTRIBUTING.md LICENSE THIRD_PARTY.md DESTINATION .)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "VRCVideoProxy-${PROJECT_VERSION}-windows-x64")
set(CPACK_PACKAGE_DIRECTORY "${CMAKE_BINARY_DIR}/dist")
include(CPack)
